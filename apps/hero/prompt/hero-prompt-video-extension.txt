        // Tailwind palette color definitions
        const colorShades = ["50", "100", "200", "300", "400", "500", "600", "700", "800", "900", "950"];
        const colorFamilies = [
            "slate", "gray", "zinc", "neutral", "stone",
            "red", "orange", "amber", "yellow", "lime",
            "green", "emerald", "teal", "cyan", "sky",
            "blue", "indigo", "violet", "purple", "fuchsia",
            "pink", "rose"
        ];

        let gradientEverEnabled = false;

        // Store settings including active target, selected colours, overlay opacity, and gradient configuration.
        const settings = {
            centreText: false,
            minimumHeight: true,
            showSubheading: true,
            gradientOverlay: false,
            overlayOpacity: 0.4,
            h1Serif: false,
            h1LetterSpacing: -0.05,
            h2Serif: false,
            h2LetterSpacing: -0.025,
            activeTarget: "overlay", // "overlay", "start", "end", "h1", or "h2"
            overlayColor: "black",
            startColor: "black",
            endColor: "blue-500",
            gradientDirection: "to right",
            h1Color: "white",
            h2Color: "white"
        };

        function readSettings() {
            settings.centreText = centreCheckbox.checked;
            settings.minimumHeight = heightCheckbox.checked;
            settings.showSubheading = subheadingCheckbox.checked;
            settings.gradientOverlay = gradientCheckbox.checked;
            settings.overlayOpacity = parseFloat(opacityRange.value);
            settings.h1Serif = h1SerifCheckbox.checked;
            settings.h1LetterSpacing = parseFloat(h1SpacingRange.value);
            settings.h2Serif = h2SerifCheckbox.checked;
            settings.h2LetterSpacing = parseFloat(h2SpacingRange.value);
            settings.gradientDirection = gradientDirectionSelect.value;
        }

        // Build the colour palette table rows
        function buildPaletteUI() {
            paletteTbody.innerHTML = "";

            // Special Row for Black & White
            const specialTr = document.createElement("tr");
            specialTr.className = "border-b border-slate-200/80";

            const specialLabelTd = document.createElement("td");
            specialLabelTd.className = "pr-3 py-1 font-medium text-slate-600 text-xs capitalize";
            specialLabelTd.textContent = "Monochrome";
            specialTr.appendChild(specialLabelTd);

            const whiteTd = document.createElement("td");
            whiteTd.className = "p-1 text-center";
            whiteTd.appendChild(createSwatchButton("white"));
            specialTr.appendChild(whiteTd);

            const blackTd = document.createElement("td");
            blackTd.className = "p-1 text-center";
            blackTd.appendChild(createSwatchButton("black"));
            specialTr.appendChild(blackTd);

            // Empty cells for remaining columns in monochrome row
            for (let i = 2; i < 11; i++) {
                const emptyTd = document.createElement("td");
                emptyTd.className = "p-1";
                specialTr.appendChild(emptyTd);
            }
            paletteTbody.appendChild(specialTr);

            // Standard Tailwind Colour Family Rows
            colorFamilies.forEach(function (family) {
                const tr = document.createElement("tr");
                tr.className = "hover:bg-slate-100/50 transition-colors";

                const labelTd = document.createElement("td");
                labelTd.className = "pr-3 py-1 font-medium text-slate-600 text-xs capitalize";
                labelTd.textContent = family;
                tr.appendChild(labelTd);

                colorShades.forEach(function (shade) {
                    const td = document.createElement("td");
                    td.className = "p-1 text-center";

                    const colorName = `${family}-${shade}`;
                    const button = createSwatchButton(colorName);

                    td.appendChild(button);
                    tr.appendChild(td);
                });

                paletteTbody.appendChild(tr);
            });
        }

        function createSwatchButton(colorName) {
            const button = document.createElement("button");
            button.type = "button";
            button.className = "swatch-btn";
            button.dataset.color = colorName;
            button.setAttribute("aria-label", `Select colour ${colorName}`);
            button.title = colorName;

            // Apply Tailwind variable style directly to swatch for accurate background representation
            button.style.backgroundColor = `var(--color-${colorName})`;

            button.addEventListener("click", function () {
                selectColorForActiveTarget(colorName);
            });

            return button;
        }

        function selectColorForActiveTarget(colorName) {
            if (settings.activeTarget === "overlay") {
                settings.overlayColor = colorName;
            } else if (settings.activeTarget === "start") {
                settings.startColor = colorName;
            } else if (settings.activeTarget === "end") {
                settings.endColor = colorName;
            } else if (settings.activeTarget === "h1") {
                settings.h1Color = colorName;
            } else if (settings.activeTarget === "h2") {
                settings.h2Color = colorName;
            }
            updateHero();
        }

        function setActiveTarget(target) {
            // Validate target against active state/mode fallbacks
            if (target === "h2" && !settings.showSubheading) {
                target = settings.gradientOverlay ? "start" : "overlay";
            }
            if (!settings.gradientOverlay && (target === "start" || target === "end")) {
                target = "overlay";
            }
            if (settings.gradientOverlay && target === "overlay") {
                target = "start";
            }

            settings.activeTarget = target;

            const targetBtns = [targetOverlayBtn, targetStartBtn, targetEndBtn, targetH1Btn, targetH2Btn];
            targetBtns.forEach(function (btn) {
                btn.classList.remove("active");
                btn.setAttribute("aria-selected", "false");
            });

            if (target === "overlay") {
                targetOverlayBtn.classList.add("active");
                targetOverlayBtn.setAttribute("aria-selected", "true");
            } else if (target === "start") {
                targetStartBtn.classList.add("active");
                targetStartBtn.setAttribute("aria-selected", "true");
            } else if (target === "end") {
                targetEndBtn.classList.add("active");
                targetEndBtn.setAttribute("aria-selected", "true");
            } else if (target === "h1") {
                targetH1Btn.classList.add("active");
                targetH1Btn.setAttribute("aria-selected", "true");
            } else if (target === "h2") {
                targetH2Btn.classList.add("active");
                targetH2Btn.setAttribute("aria-selected", "true");
            }

            updateSwatchHighlights();
        }

        function updateSwatchHighlights() {
            let activeColor = settings.overlayColor;
            if (settings.activeTarget === "start") activeColor = settings.startColor;
            if (settings.activeTarget === "end") activeColor = settings.endColor;
            if (settings.activeTarget === "h1") activeColor = settings.h1Color;
            if (settings.activeTarget === "h2") activeColor = settings.h2Color;

            const swatches = paletteTbody.querySelectorAll(".swatch-btn");
            swatches.forEach(function (swatch) {
                if (swatch.dataset.color === activeColor) {
                    swatch.classList.add("selected-swatch");
                } else {
                    swatch.classList.remove("selected-swatch");
                }
            });
        }

        // Use the same class choices for preview and export.
        function getHeroClasses() {
            const classes = ["hero"];

            if (settings.centreText) {
                classes.push("hero-centre");
            }

            if (settings.minimumHeight) {
                classes.push("hero-min-height");
            }

            if (settings.h1Serif) {
                classes.push("hero-h1-serif");
            }

            if (settings.h2Serif) {
                classes.push("hero-h2-serif");
            }

            return classes;
        }

        function handleGradientToggleChange() {
            const isGradient = gradientCheckbox.checked;

            if (isGradient && !gradientEverEnabled) {
                // First enabling: use current solid overlay colour as start, blue-500 as end
                settings.startColor = settings.overlayColor;
                settings.endColor = "blue-500";
                gradientEverEnabled = true;
            }

            if (isGradient) {
                targetOverlayBtn.hidden = true;
                targetStartBtn.hidden = false;
                targetEndBtn.hidden = false;
                gradientDirectionWrapper.hidden = false;

                if (settings.activeTarget === "overlay") {
                    setActiveTarget("start");
                }
            } else {
                targetOverlayBtn.hidden = false;
                targetStartBtn.hidden = true;
                targetEndBtn.hidden = true;
                gradientDirectionWrapper.hidden = true;

                if (settings.activeTarget === "start" || settings.activeTarget === "end") {
                    setActiveTarget("overlay");
                }
            }

            updateHero();
        }

        function buildDynamicHeroCss(imagePath) {
            let overlayStyleRules = "";
            if (settings.gradientOverlay) {
                overlayStyleRules = `
                    background-color: transparent;
                    background-image: linear-gradient(${settings.gradientDirection}, var(--color-${settings.startColor}), var(--color-${settings.endColor}));
                    opacity: ${settings.overlayOpacity};
                `;
            } else {
                overlayStyleRules = `
                    background-color: var(--color-${settings.overlayColor});
                    background-image: none;
                    opacity: ${settings.overlayOpacity};
                `;
            }

            return `
                #hero-preview {
                    background-image: url('${imagePath}');
                }
                #hero-overlay {
                    ${overlayStyleRules}
                }
                #hero-heading {
                    color: var(--color-${settings.h1Color});
                    letter-spacing: ${settings.h1LetterSpacing}em;
                }
                #hero-subheading {
                    color: var(--color-${settings.h2Color});
                    letter-spacing: ${settings.h2LetterSpacing}em;
                }
            `;
        }

        function updateHero() {
            readSettings();

            const activeClasses = getHeroClasses();

            const optionalClasses = [
                "hero-centre",
                "hero-min-height",
                "hero-h1-serif",
                "hero-h2-serif"
            ];

            optionalClasses.forEach(function (className) {
                heroPreview.classList.toggle(
                    className,
                    activeClasses.includes(className)
                );
            });

            // Update Target Button Labels
            targetOverlayVal.textContent = settings.overlayColor;
            targetStartVal.textContent = settings.startColor;
            targetEndVal.textContent = settings.endColor;
            targetH1Val.textContent = settings.h1Color;
            targetH2Val.textContent = settings.h2Color;

            // Handle H2 state changes and target fallbacks
            if (!settings.showSubheading) {
                targetH2Btn.disabled = true;
                if (settings.activeTarget === "h2") {
                    setActiveTarget(settings.gradientOverlay ? "start" : "overlay");
                }
            } else {
                targetH2Btn.disabled = false;
            }

            // Ensure correct active target focus
            setActiveTarget(settings.activeTarget);

            // Update dynamic colors, overlay rules, and letter spacing styles for app preview
            dynamicHeroStyles.textContent = buildDynamicHeroCss(appImagePath);

            // Disable/enable H2 effect controls when subheading is hidden
            h2SerifCheckbox.disabled = !settings.showSubheading;
            h2SpacingRange.disabled = !settings.showSubheading;

            if (!settings.showSubheading) {
                h2ControlsGroup.classList.add("opacity-50", "cursor-not-allowed");
            } else {
                h2ControlsGroup.classList.remove("opacity-50", "cursor-not-allowed");
            }

            // Preserve the H2 element and its text while hiding it.
            heroSubheading.hidden = !settings.showSubheading;

            updateSwatchHighlights();
            clearStatus();
            updateManualCode();
        }

        // Keep special characters as text in the exported document.
        function escapeHtml(text) {
            return text
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#39;");
        }

        function generateHtml() {
            readSettings();

            const headingText = escapeHtml(heroHeading.textContent);
            const subheadingText = escapeHtml(heroSubheading.textContent);
            const classNames = getHeroClasses().join(" ");
            const sharedCss = heroStyles.textContent.trim();
            const dynamicCss = buildDynamicHeroCss(exportedImagePath).trim();

            let subheadingHtml = "";

            if (settings.showSubheading) {
                subheadingHtml = `
            <h2 id="hero-subheading">${subheadingText}</h2>`;
            }

            // Escape the closing script tag inside this string.
            return `<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <meta name="description" content="Short, unique description of web page. Often used as the snippet shown in search results. About 120 characters is ideal.">

    <title>Hero Page</title>

    <link rel="preload" as="image" href="${exportedImagePath}">

    <script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>

    <style type="text/tailwindcss">
${sharedCss}
    </style>
    <style type="text/tailwindcss">
${dynamicCss}
    </style>
</head>

<body>
    <header id="hero-preview" class="${classNames}">
        <div id="hero-overlay" class="hero-overlay" aria-hidden="true"></div>
        <div class="hero-content">
            <h1 id="hero-heading">${headingText}</h1>${subheadingHtml}
        </div>
    </header>
</body>
</html>`;
        }

        function clearStatus() {
            copyStatus.textContent = "";
        }

        function updateManualCode() {
            if (!manualCopy.hidden) {
                manualCode.value = generateHtml();
            }
        }

        function handleHeadingInput() {
            clearStatus();
            updateManualCode();
        }

        // Insert plain text without creating HTML elements.
        function insertPlainText(heading, text) {
            const selection = window.getSelection();

            if (!selection || selection.rangeCount === 0) {
                return;
            }

            const range = selection.getRangeAt(0);

            if (
                !heading.contains(range.startContainer) ||
                !heading.contains(range.endContainer)
            ) {
                return;
            }

            range.deleteContents();

            const textNode = document.createTextNode(text);
            range.insertNode(textNode);

            range.setStartAfter(textNode);
            range.collapse(true);

            selection.removeAllRanges();
            selection.addRange(range);

            handleHeadingInput();
        }

        function handlePlainTextPaste(event) {
            event.preventDefault();

            if (!event.clipboardData) {
                return;
            }

            const pastedText = event.clipboardData.getData("text/plain");

            insertPlainText(event.currentTarget, pastedText);
        }

        // Prevent Enter from creating extra HTML elements.
        function handleHeadingBeforeInput(event) {
            if (
                event.inputType === "insertParagraph" ||
                event.inputType === "insertLineBreak"
            ) {
                event.preventDefault();
                insertPlainText(event.currentTarget, "\n");
            }
        }

        function preventFormattedDrop(event) {
            event.preventDefault();
        }

        function showManualCopy(html) {
            manualCopy.hidden = false;
            manualCode.value = html;

            copyStatus.textContent =
                "Automatic copying is unavailable. Use the generated HTML below.";

            manualCode.focus();
            manualCode.select();
        }

        async function copyHtml() {
            clearStatus();

            const html = generateHtml();

            try {
                if (!navigator.clipboard || !navigator.clipboard.writeText) {
                    throw new Error("Clipboard API is unavailable.");
                }

                await navigator.clipboard.writeText(html);

                copyStatus.textContent =
                    "HTML copied—paste it into VS Code.";
            } catch (error) {
                showManualCopy(generateHtml());
            }
        }

        // Initialize target button handlers
        targetOverlayBtn.addEventListener("click", function () { setActiveTarget("overlay"); });
        targetStartBtn.addEventListener("click", function () { setActiveTarget("start"); });
        targetEndBtn.addEventListener("click", function () { setActiveTarget("end"); });
        targetH1Btn.addEventListener("click", function () { setActiveTarget("h1"); });
        targetH2Btn.addEventListener("click", function () { setActiveTarget("h2"); });

        // Gradient Mode Checkbox & Direction Listeners
        gradientCheckbox.addEventListener("change", handleGradientToggleChange);
        gradientDirectionSelect.addEventListener("change", updateHero);

        // Opacity Slider Listener
        opacityRange.addEventListener("input", function () {
            opacityVal.textContent = opacityRange.value;
            updateHero();
        });

        // Update preview when standard checkboxes change
        const checkboxes = [
            centreCheckbox,
            heightCheckbox,
            subheadingCheckbox,
            h1SerifCheckbox,
            h2SerifCheckbox
        ];

        checkboxes.forEach(function (checkbox) {
            checkbox.addEventListener("change", updateHero);
        });

        // Update letter spacing immediately while dragging sliders
        h1SpacingRange.addEventListener("input", function () {
            h1SpacingVal.textContent = h1SpacingRange.value + "em";
            updateHero();
        });

        h2SpacingRange.addEventListener("input", function () {
            h2SpacingVal.textContent = h2SpacingRange.value + "em";
            updateHero();
        });

        // Keep editing plain text and the manual HTML current.
        const editableHeadings = [heroHeading, heroSubheading];

        editableHeadings.forEach(function (heading) {
            heading.addEventListener("input", handleHeadingInput);
            heading.addEventListener("paste", handlePlainTextPaste);
            heading.addEventListener(
                "beforeinput",
                handleHeadingBeforeInput
            );
            heading.addEventListener("drop", preventFormattedDrop);
        });

        copyButton.addEventListener("click", copyHtml);

        // Build Palette and Apply Initial State
        buildPaletteUI();
        updateHero();
    </script>
</body>

</html>
